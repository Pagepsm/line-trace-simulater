function(verify_dependency name revision)
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${CMAKE_CURRENT_SOURCE_DIR}/vendor/${name}" rev-parse HEAD
    OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  if(NOT actual STREQUAL revision)
    message(FATAL_ERROR "${name}: expected ${revision}, got ${actual}")
  endif()
endfunction()
